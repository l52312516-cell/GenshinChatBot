cmake_minimum_required(VERSION 3.24)
project(ChatGIBotNative LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(ChatGIBotNative WIN32 resources.rc src/main.c src/ui.c src/util.c src/config.c src/dependencies.c src/capture.c src/ai.c src/ocr.c src/vision.c src/bot.c src/plugin_manager.c src/overlay.c src/gdi_ui.cpp third_party/cjson/cJSON.c)
target_compile_definitions(ChatGIBotNative PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
target_include_directories(ChatGIBotNative PRIVATE src third_party/cjson)
target_link_libraries(ChatGIBotNative PRIVATE gdi32 gdiplus dwmapi d3d11 uxtheme comctl32 shlwapi winhttp bcrypt dxgi ole32 windowscodecs comdlg32 shell32)
if(MSVC)
    target_compile_options(ChatGIBotNative PRIVATE /O2 /W4 /utf-8 /GS)
    target_link_options(ChatGIBotNative PRIVATE /SUBSYSTEM:WINDOWS /DYNAMICBASE /NXCOMPAT)
else()
    target_compile_options(ChatGIBotNative PRIVATE -O2 -Wall -Wextra)
    target_link_options(ChatGIBotNative PRIVATE -mwindows)
endif()
